# 第四章 系统实现

## 4.1 概述

本章将详细介绍在线考试系统的具体实现过程，包括项目结构、核心模块实现、数据库实现、安全机制实现等方面。系统采用TypeScript + Node.js + Koa的技术栈，通过模块化设计和分层架构，实现了高性能、高安全性的在线考试系统。

### 4.2.2 模块职责划分

1. **Controller层**：处理HTTP请求，调用Service层业务逻辑
2. **Service层**：实现核心业务逻辑，处理数据验证和业务规则
3. **Repository层**：数据访问抽象，封装数据库操作
4. **Entity层**：数据库实体定义，映射数据库表结构
5. **DTO层**：数据传输对象，定义API接口数据结构
6. **Middleware层**：中间件功能，处理横切关注点
7. **Util层**：工具类，提供通用功能支持

```typescript
  interval: 1000

app.use(authMiddleware);            // 认证中间件
app.use(antiCrawlerMiddleware);     // 防爬虫中间件
app.use(router.routes()).use(router.allowedMethods());

// 全局错误处理
app.on('error', (err, ctx) => {
  logger().error({
    event: "serverError",
    message: "未捕获的服务器错误",
    error: err,
  });
  
  if (!ctx.res.headersSent) {
    ctx.status = 500;
    ctx.body = {
      success: false,
      code: 500,
      error: {
        code: 'INTERNAL_SERVER_ERROR',
        message: '服务器内部错误',
        timestamp: new Date().toISOString()
      }
    };
  }
});

// 优雅关闭
const gracefulShutdown = (signal: string) => {
  logger().info({ event: "shutdown", message: signal });
  process.exit(0);
};

process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT', () => gracefulShutdown('SIGINT'));

export default app;
```

### 4.3.2 数据访问层实现

**BaseRepo.ts - 基础仓库类**

基础仓库类提供通用的数据访问方法：

```typescript
import { Repository, FindManyOptions, FindOneOptions } from "typeorm";
import { AppDataSource } from "../util/orm";

export abstract class BaseRepo<T> {
  protected repository: Repository<T>;

  constructor(entity: new () => T) {
    this.repository = AppDataSource.getRepository(entity);
  }

  async findAll(options?: FindManyOptions<T>): Promise<T[]> {
    return await this.repository.find(options);
  }

  async findOne(options: FindOneOptions<T>): Promise<T | null> {
    return await this.repository.findOne(options);
  }

  async findById(id: number): Promise<T | null> {
    return await this.repository.findOne({ where: { id } as any });
  }

  async create(data: Partial<T>): Promise<T> {
    const entity = this.repository.create(data);
    return await this.repository.save(entity);
  }

  async update(id: number, data: Partial<T>): Promise<T | null> {
    await this.repository.update(id, data);
    return await this.findById(id);
  }

  async delete(id: number): Promise<boolean> {
    const result = await this.repository.delete(id);
    return result.affected !== 0;
  }

  async count(options?: FindManyOptions<T>): Promise<number> {
    return await this.repository.count(options);
  }
}
```

### 4.3.3 业务逻辑层实现

**ExamService - 考试服务实现**

考试服务实现核心的考试业务逻辑：

```typescript
import { Repository } from "typeorm";
import { AppDataSource } from "../util/orm";
import { Exam, Question, ExamQuestion, ExamAnswer } from "../orm/entity";
import { CreateExamDto, UpdateExamDto, ExamSubmitDto } from "../dto/exam";
import { logger } from "../util/log";

export class ExamService {
  private examRepository: Repository<Exam>;
  private questionRepository: Repository<Question>;
  private examQuestionRepository: Repository<ExamQuestion>;
  private examAnswerRepository: Repository<ExamAnswer>;

  constructor() {
    this.examRepository = AppDataSource.getRepository(Exam);
    this.questionRepository = AppDataSource.getRepository(Question);
    this.examQuestionRepository = AppDataSource.getRepository(ExamQuestion);
    this.examAnswerRepository = AppDataSource.getRepository(ExamAnswer);
  }

  async createExam(createExamDto: CreateExamDto, userId: number): Promise<Exam> {
    try {
      const exam = this.examRepository.create({
        ...createExamDto,
        createdBy: userId,
        isPublished: false
      });

      const savedExam = await this.examRepository.save(exam);
      
      logger().info({
        event: "exam_created",
        message: `考试创建成功: ${savedExam.id}`,
        userId,
        examId: savedExam.id
      });

      return savedExam;
    } catch (error) {
      logger().error({
        event: "exam_create_error",
        message: "考试创建失败",
        error,
        userId
      });
      throw error;
    }
  }

  async updateExam(id: number, updateExamDto: UpdateExamDto, userId: number): Promise<Exam | null> {
    try {
      const exam = await this.examRepository.findOne({ where: { id } });
      if (!exam) {
        throw new Error("考试不存在");
      }

      await this.examRepository.update(id, {
        ...updateExamDto,
        updatedBy: userId
      });

      const updatedExam = await this.examRepository.findOne({ where: { id } });
      
      logger().info({
        event: "exam_updated",
        message: `考试更新成功: ${id}`,
        userId,
        examId: id
      });

      return updatedExam;
    } catch (error) {
      logger().error({
        event: "exam_update_error",
        message: "考试更新失败",
        error,
        userId,
        examId: id
      });
      throw error;
    }
  }

  async getExamList(page: number = 1, limit: number = 10, userId?: number): Promise<{ exams: Exam[], total: number }> {
    try {
      const queryBuilder = this.examRepository.createQueryBuilder("exam")
        .leftJoinAndSelect("exam.createdBy", "creator")
        .orderBy("exam.createdAt", "DESC");

      if (userId) {
        queryBuilder.where("exam.createdBy = :userId", { userId });
      }

      const [exams, total] = await queryBuilder
        .skip((page - 1) * limit)
        .take(limit)
        .getManyAndCount();

      return { exams, total };
    } catch (error) {
      logger().error({
        event: "exam_list_error",
        message: "获取考试列表失败",
        error,
        userId
      });
      throw error;
    }
  }

  async getExamDetail(id: number): Promise<Exam | null> {
    try {
      const exam = await this.examRepository.findOne({
        where: { id },
        relations: ["questions", "questions.question"]
      });

      return exam;
    } catch (error) {
      logger().error({
        event: "exam_detail_error",
        message: "获取考试详情失败",
        error,
        examId: id
      });
      throw error;
    }
  }

  async submitExam(examSubmitDto: ExamSubmitDto, userId: number): Promise<void> {
    try {
      const { examId, answers } = examSubmitDto;

      // 验证考试是否存在
      const exam = await this.examRepository.findOne({ where: { id: examId } });
      if (!exam) {
        throw new Error("考试不存在");
      }

      // 保存答案
      for (const answer of answers) {
        const examAnswer = this.examAnswerRepository.create({
          examId,
          questionId: answer.questionId,
          userId,
          answer: answer.answer,
          score: 0, // 初始分数为0，需要后续批改
          isCorrect: false
        });

        await this.examAnswerRepository.save(examAnswer);
      }

      logger().info({
        event: "exam_submitted",
        message: `考试提交成功: ${examId}`,
        userId,
        examId
      });
    } catch (error) {
      logger().error({
        event: "exam_submit_error",
        message: "考试提交失败",
        error,
        userId,
        examId: examSubmitDto.examId
      });
      throw error;
    }
  }

  async getExamStatistics(examId: number): Promise<any> {
    try {
      const exam = await this.examRepository.findOne({ where: { id: examId } });
      if (!exam) {
        throw new Error("考试不存在");
      }

      const totalAnswers = await this.examAnswerRepository.count({
        where: { examId }
      });

      const correctAnswers = await this.examAnswerRepository.count({
        where: { examId, isCorrect: true }
      });

      const averageScore = await this.examAnswerRepository
        .createQueryBuilder("answer")
        .select("AVG(answer.score)", "avgScore")
        .where("answer.examId = :examId", { examId })
        .getRawOne();

      return {
        examId,
        totalAnswers,
        correctAnswers,
        accuracy: totalAnswers > 0 ? (correctAnswers / totalAnswers * 100).toFixed(2) : 0,
        averageScore: averageScore.avgScore || 0
      };
    } catch (error) {
      logger().error({
        event: "exam_statistics_error",
        message: "获取考试统计失败",
        error,
        examId
      });
      throw error;
    }
  }
}

export const examService = new ExamService();
```

## 4.4 中间件实现

### 4.4.1 认证中间件

**auth.middleware.ts - JWT认证中间件**

```typescript
import { Context, Next } from "koa";
import { ServerConfig } from "../util/env";
import { verifyToken, whiteListRegex } from "../util/crypto";
import { getRedisService } from "../util/redis";
import { UnauthorizedError } from "../util/error";

const authMiddleware = async (ctx: Context, next: Next) => {
  const path = ctx.request.path;
  
  // 检查白名单
  if (
    ServerConfig.jwt.whiteList.includes(path) ||
    (await whiteListRegex(path))
  ) {
    await next();
    return;
  }

  // 获取token
  const token = ctx.request.headers.authorization;
  if (!token) throw new UnauthorizedError("缺少token");

  // 检查黑名单
  const redisService = getRedisService();
  const blackToken = `${ServerConfig.jwt.blackListPrefix}${token}`;
  if (await redisService.existCache(blackToken))
    throw new UnauthorizedError("该登录状态已被抛弃,请重新登陆获取状态");

  // 登出时加入黑名单
  if (ctx.request.path === "/logout")
    await redisService.setCacheEx(blackToken, "1", 60 * 60 * 24);

  // 验证token
  const replaceToken = token.replace("Bearer ", "");
  const account = await verifyToken(replaceToken);
  ctx.state.account = account;
  
  await next();
};

export default authMiddleware;
```

```typescript
  const user = ctx.state.user;
  if (user) {
    const subject = user?.role || "base";
    const object = ctx.path;
    const action = ctx.method.toLowerCase();
    const pass = await casbinService.enforce(subject, object, action);
```

### 4.4.3 限流中间件

**limit.middleware.ts - 基于令牌桶的限流**

```typescript
import { Context, Next } from "koa";
import { ManyRequestError } from "../util/error";

interface limitOption {
  tokensPerInterval: number;
  interval: number;
}

const rateLimitMap = new Map<string, { tokens: number; last: number }>();

export function rateLimitMiddleware(options: limitOption) {
  return async (ctx: Context, next: Next) => {
    const key = ctx.ip;
    const now = Date.now();
    const bucket = rateLimitMap.get(key) || {
      tokens: options.tokensPerInterval,
      last: now,
    };

    const elapsed = now - bucket.last;
    const refill = Math.floor(elapsed / options.interval);
    if (refill > 0) {
      bucket.tokens = Math.min(
        options.tokensPerInterval,
        bucket.tokens + refill
      );
      bucket.last = now;
    }

    if (bucket.tokens > 0) {
      bucket.tokens -= 1;
      rateLimitMap.set(key, bucket);
      await next();
      return;
    }
    throw new ManyRequestError("Too Many Requests");
  };
}
```

### 4.4.4 响应格式化中间件

**response.middleware.ts - 统一响应格式**

```typescript
import { Context, Next } from "koa";
import {
  CustomError,
  FireError,
  ForbiddenError,
  ManyRequestError,
  UnauthorizedError,
  ValidateParamsError,
} from "../util/error";
import { logger } from "../util/log";
import { LogRepository } from "../orm/repository/exam";
import { logData } from "../util/key";

export async function responseFormatter(ctx: Context, next: Next) {
  let logData: logData = {
    level: "",
    message: "",
    role: "",
    path: ctx.path,
    method: ctx.method,
    userId: "",
    resStatus: 0,
    traceId: "",
  };

  try {
    await next();
    
    // 是否跳过格式化（导出文件等场景）
    const skip = ctx.state.skipFormat || Buffer.isBuffer(ctx.body);
    if (!skip) {
      // 成功响应格式化
      if (ctx.body !== undefined && ctx.status === 200) {
        ctx.body = {
          code: 0,
          message: "success",
          data: ctx.body,
          success: true,
          timestamp: new Date().toLocaleString(),
          traceId: ctx.state.traceId,
        };
      } else {
        // 异常状态码处理
        logger().warn({
          event: "请求路由异常状态码捕捉",
          message: String(ctx.status),
        });
        ctx.status = ctx.status;
        ctx.body = {
          code: -1,
          message: "请求异常,请检查后重试",
          data: null,
          success: false,
          timestamp: new Date().toLocaleString(),
          traceId: ctx.state.traceId,
        };
      }

      // 记录成功日志
      logger().info({
        event: "响应日志",
        message: "success",
      });
      
      logData.level = "info";
      logData.message = "请求成功";
      logData.role = ctx.state.account?.role || "游客登录";
      logData.userId = ctx.state.account?.id || 0;
      logData.resStatus = ctx.status;
      logData.traceId = ctx.state.traceId;
      await LogRepository.create(logData);
    }
  } catch (err: any) {
    // 错误处理
    if (err instanceof CustomError) {
      ctx.status = err.statusCode;
      ctx.body = {
        code: err.statusCode,
        message: err.message,
        data: null,
        success: false,
        timestamp: new Date().toLocaleString(),
        traceId: ctx.state.traceId,
        error: {
          code: err.code,
          message: err.message,
        },
      };
    } else {
      // 未知错误
      ctx.status = 500;
      ctx.body = {
        code: 500,
        message: "服务器内部错误",
        data: null,
        success: false,
        timestamp: new Date().toLocaleString(),
        traceId: ctx.state.traceId,
        error: {
          code: "INTERNAL_SERVER_ERROR",
          message: err.message || "未知错误",
        },
      };
    }

    // 记录错误日志
    logger().error({
      event: "响应错误日志",
      message: err.message,
      error: err,
    });

    logData.level = "error";
    logData.message = err.message || "未知错误";
    logData.role = ctx.state.account?.role || "游客登录";
    logData.userId = ctx.state.account?.id || 0;
    logData.resStatus = ctx.status;
    logData.traceId = ctx.state.traceId;
    await LogRepository.create(logData);
  }
}
```

```typescript
import bcrypt from "bcrypt";
import jwt from "jsonwebtoken";
import crypto from "crypto";
import { ServerConfig } from "./env";

export class CryptoUtil {
  // 密码加密
  static async hashPassword(password: string): Promise<{ hash: string; salt: string }> {
    const salt = await bcrypt.genSalt(10);
    const hash = await bcrypt.hash(password, salt);
    return { hash, salt };
  }

  // 密码验证
  static async verifyPassword(password: string, hash: string): Promise<boolean> {
    return await bcrypt.compare(password, hash);
  }

  // 生成JWT令牌
  static generateToken(payload: any): string {
    return jwt.sign(payload, ServerConfig.jwt.secret, {
      expiresIn: ServerConfig.jwt.expiresIn,
    });
  }

  // 验证JWT令牌
  static verifyToken(token: string): any {
    return jwt.verify(token, ServerConfig.jwt.secret);
  }

  // 生成随机字符串
  static generateRandomString(length: number = 32): string {
    return crypto.randomBytes(length).toString('hex');
  }

  // 生成UUID
  static generateUUID(): string {
    return crypto.randomUUID();
  }
}

export const { hashPassword, verifyPassword, generateToken, verifyToken, generateRandomString, generateUUID } = CryptoUtil;
```

## 4.6 WebSocket实现

### 4.6.1 WebSocket服务

**ws/index.ts - WebSocket服务初始化**

```typescript
import { Server } from "http";
import { WebSocketServer, WebSocket } from "ws";
import { logger } from "../util/log";
import { verifyToken } from "../util/crypto";
import { ConnectionPool } from "./pool";
import { MessageHandler } from "./message";

export function initWebSocket(server: Server) {
  const wss = new WebSocketServer({ server });
  const connectionPool = new ConnectionPool();
  const messageHandler = new MessageHandler(connectionPool);

  wss.on('connection', async (ws: WebSocket, request) => {
    try {
      // 从查询参数获取token
      const url = new URL(request.url!, `http://${request.headers.host}`);
      const token = url.searchParams.get('token');

      if (!token) {
        ws.close(1008, 'Authentication required');
        return;
      }

      // 验证token
      const user = verifyToken(token);
      if (!user) {
        ws.close(1008, 'Invalid token');
        return;
      }

      // 添加到连接池
      connectionPool.addConnection(user.id, ws);

      logger().info({
        event: "websocket_connected",
        message: `用户 ${user.id} 连接WebSocket`,
        userId: user.id
      });

      // 发送欢迎消息
      ws.send(JSON.stringify({
        type: 'welcome',
        data: {
          message: '连接成功',
          userId: user.id
        }
      }));

      // 处理消息
      ws.on('message', (data) => {
        try {
          const message = JSON.parse(data.toString());
          messageHandler.handleMessage(user.id, message, ws);
        } catch (error) {
          logger().error({
            event: "websocket_message_error",
            message: "WebSocket消息处理错误",
            error,
            userId: user.id
          });
        }
      });

      // 处理断开连接
      ws.on('close', () => {
        connectionPool.removeConnection(user.id);
        logger().info({
          event: "websocket_disconnected",
          message: `用户 ${user.id} 断开WebSocket连接`,
          userId: user.id
        });
      });

      // 处理错误
      ws.on('error', (error) => {
        logger().error({
          event: "websocket_error",
          message: "WebSocket连接错误",
          error,
          userId: user.id
        });
        connectionPool.removeConnection(user.id);
      });

    } catch (error) {
      logger().error({
        event: "websocket_connection_error",
        message: "WebSocket连接初始化错误",
        error
      });
      ws.close(1008, 'Connection initialization failed');
    }
  });

  logger().info({
    event: "websocket_server_started",
    message: "WebSocket服务器启动成功"
  });
}
```

### 4.6.2 连接池管理

**ws/pool.ts - WebSocket连接池**

```typescript
import { WebSocket } from "ws";

export class ConnectionPool {
  private connections: Map<number, WebSocket> = new Map();

  addConnection(userId: number, ws: WebSocket): void {
    // 如果用户已有连接，关闭旧连接
    const existingConnection = this.connections.get(userId);
    if (existingConnection) {
      existingConnection.close();
    }
    
    this.connections.set(userId, ws);
  }

  removeConnection(userId: number): void {
    this.connections.delete(userId);
  }

  getConnection(userId: number): WebSocket | undefined {
    return this.connections.get(userId);
  }

  sendToUser(userId: number, message: any): boolean {
    const connection = this.connections.get(userId);
    if (connection && connection.readyState === WebSocket.OPEN) {
      connection.send(JSON.stringify(message));
      return true;
    }
    return false;
  }

  broadcast(message: any): void {
    this.connections.forEach((ws) => {
      if (ws.readyState === WebSocket.OPEN) {
        ws.send(JSON.stringify(message));
      }
    });
  }

  getConnectionCount(): number {
    return this.connections.size;
  }

  getAllUserIds(): number[] {
    return Array.from(this.connections.keys());
  }
}
```
